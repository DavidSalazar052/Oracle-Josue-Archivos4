CREATE OR REPLACE TRIGGER trg_empleado
AFTER INSERT ON empleados
BEGIN
    DBMS_OUTPUT.PUT_LINE('Empleado insertado');
END;
/

CREATE OR REPLACE TRIGGER trg_validar_salario
BEFORE INSERT OR UPDATE ON empleados
FOR EACH ROW
BEGIN

    IF :NEW.salario < 0 THEN
        RAISE_APPLICATION_ERROR(
            -20001,
            'El salario no puede ser negativo'
        );
    END IF;

END;
/

create or replace function fun_invierte_may_min
(p_texto in varchar2) return varchar2 is
begin
    if p_texto = upper(p_texto) then
        return lower(p_texto);
    elsif p_texto = lower(p_texto) then
        return upper(p_texto);
    else
        return p_texto;
    end if;
end;
/


En Oracle Database, ROLLBACK es un comando que sirve para deshacer 
cambios realizados en la base de datos que todavía no han sido confirmados.

Revierte operaciones como:

INSERT
UPDATE
DELETE

siempre que aún no hayas ejecutado:

EJEMPLO ===================================
BEGIN

    UPDATE cuentas
    SET saldo = saldo - 100
    WHERE id = 1;

    UPDATE cuentas
    SET saldo = saldo + 100
    WHERE id = 2;

    COMMIT;

EXCEPTION

    WHEN OTHERS THEN
        ROLLBACK;

END;
/

COMMIT	    Guarda cambios
ROLLBACK	Deshace cambios
SAVEPOINT	Punto de retorno
